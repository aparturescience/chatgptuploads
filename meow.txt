========== STABILITY SETUP SCRIPT ==========
#!/bin/sh

total_memory=$(grep MemTotal /proc/meminfo | awk '{print $2}')
scout_enabled=$(getprop persist.sys.stability.scout.enable)

# For all devices, enable scout by default
if [ -z $scout_enabled ]; then
    setprop persist.sys.stability.scout.enable true
fi

# For device with less than 6GB RAM, scout will work in lite mode. In LT mode
# scout will neither collet backtrace nor do any action that may cause exessive load.
scout_lightweight=$(getprop persist.sys.stability.litescout.enable)
if [ $total_memory -le 6291456 ]; then
    library_test=$(getprop persist.mtbf.test)
    omni_test=$(getprop persist.omni.test)
    sentinel_enabled=$(getprop persist.sys.stability.enable_sentinel_resource_monitor)
    # Sentinel is enabled on all devices by default in the prodcut config file
    # but on the user side, we didn't see any actual benefits on low-end devices
    # so disable it unless in MTBF/UAT to save cpu/memory/battery consumption
    if [ "$sentinel_enabled" = "true" ] && [ -z "$library_test" ]; then
        # If someone need to re-enable this feature by cloud config
        # they could set the value to 1 instead of true
        setprop persist.sys.stability.enable_sentinel_resource_monitor false
    fi
    if [ $total_memory -le 4194304 ] || [ "$library_test" != "true" ] && [ "$omni_test" != "1" ]; then
        if [ -z $scout_lightweight ]; then
            setprop persist.sys.stability.litescout.enable true
        fi
    else
        # While in MTBF/UAT, Scout will work in normal mode for 6GB RAM devices
        setprop persist.sys.stability.litescout.enable false
    fi
    # Set input anr timeout to 10s for devices with <=6G RAM
    input_anr_timeout=$(getprop persist.sys.stability.input_anr_timeout)
    if [ -z $input_anr_timeout ]; then
        setprop persist.sys.stability.input_anr_timeout 10000
    fi
fi

# === BEGIN MIUI ADD: 4GB 以下机型强制 litescout=true ===
litescout_dump_disable=$(getprop persist.litescout.dump.disable)
if [ "$litescout_dump_disable" = "true" ] && [ "$total_memory" -lt 4194304 ]; then
    setprop persist.sys.stability.litescout.enable true
fi
# === END MIUI ADD ===


# For all device, scout will detect UI frozen and report it.
scout_detect_frozen=$(getprop persist.sys.stability.scout.check_frozen)
if [ -z $scout_detect_frozen ]; then
    setprop persist.sys.stability.scout.check_frozen true
fi


========== HYPSYS INIT RC ==========
on init
    # check device update type
    setprop sys.ota.type recovery

# reset device update type when use update_engine
on init && property:ro.build.ab_update=true
    setprop sys.ota.type update_engine

# 1. system/core/rootdir/init.rc 里面on init阶段 mkdir /mnt/rescue
# 2. vendor部分 on fs 阶段 mount_all 然后 mount /mnt/rescue
# 3. 这里必须放到下一个节点, on post-fs 先进行restorecon 然后start mqsasd

# mqsasd
service hypsys_system /system_ext/bin/hypsys_system
    user root
    group system root cache log everybody product_hyperengine
    disabled
    socket mqsasd stream 0660 system system
    socket mqsasd_pr dgram 0666 system system
    # keep in sync with __scudo_default_options() in main.cpp
    setenv SCUDO_OPTIONS release_to_os_interval_ms=0:quarantine_size_mb=0:thread_local_quarantine_size_kb=0:quarantine_max_chunk_size=0:dealloc_type_mismatch=false:may_return_null=true

# post fs start hypsys system
on post-fs
   start hypsys_system

# use rescue partition
on post-fs && property:sys.ota.type=update_engine
    restorecon_recursive /mnt/rescue
    mkdir /mnt/rescue/mqsas 0775 root system
    mkdir /mnt/rescue/mqsas/dumpstateTemporary 0775 root system
    mount tmpfs tmpfs /mnt/rescue/mqsas/dumpstateTemporary nodev noexec nosuid size=200M
    restorecon_recursive /mnt/rescue/mqsas
    restorecon_recursive /mnt/rescue/mqsas/dumpstateTemporary

# use cache partition
on post-fs && property:sys.ota.type=recovery
    restorecon_recursive /cache
    mkdir /cache/mqsas 0775 root system
    mkdir /cache/mqsas/dumpstateTemporary 0775 root system
    mount tmpfs tmpfs /cache/mqsas/dumpstateTemporary nodev noexec nosuid size=200M
    restorecon_recursive /cache/mqsas
    restorecon_recursive /cache/mqsas/dumpstateTemporary


# Log Manager
on post-fs-data
    mkdir /data/miuilog 0775 system system
    mkdir /data/miuilog/bsp 0775 system system
    mkdir /data/miuilog/bsp/audio 0775 audioserver audioserver
    mkdir /data/miuilog/bsp/audio/traces 0777 audioserver audioserver

    mkdir /data/mqsas 0775 system system
    mkdir /data/miuilog/stability 0775 system system
    mkdir /data/mqsas/aging 0770 system system

    mkdir /data/system/bootlog 2771 root log
    mkdir /data/system/ota_exception 2771 root log
# umount memory partition when boot monitor completed!
on property:sys.boot_monitor_completed=1 && property:sys.ota.type=update_engine
    umount /mnt/rescue/mqsas/dumpstateTemporary

# umount memory partition when boot monitor completed!
on property:sys.boot_monitor_completed=1 && property:sys.ota.type=recovery
    umount /cache/mqsas/dumpstateTemporary
